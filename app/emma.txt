        // Price Formatter based on active currency
        function formatPrice(priceInUSD) {
            const curr = currencyRates[currentCurrency];
            const converted = priceInUSD * curr.rate;
            return `${curr.symbol}${converted.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
        }

        // Apply Catalog Filters
        function applyFilters() {
            let filtered = [...products];

            if (currentCategory !== 'all') {
                filtered = filtered.filter(p => p.category === currentCategory);
            }

            const searchVal = document.getElementById('searchInput').value.toLowerCase().trim();
            if (searchVal) {
                filtered = filtered.filter(p => p.title.toLowerCase().includes(searchVal) || p.brand.toLowerCase().includes(searchVal));
            }

            const selectedBrands = Array.from(document.querySelectorAll('.brand-checkbox:checked')).map(cb => cb.value);
            if (selectedBrands.length > 0) {
                filtered = filtered.filter(p => selectedBrands.includes(p.brand));
            }

            const maxP = parseFloat(document.getElementById('priceRange').value);
            filtered = filtered.filter(p => p.priceUSD <= maxP);

            const sortVal = document.getElementById('sortBy').value;
            if (sortVal === 'lowHigh') filtered.sort((a,b) => a.priceUSD - b.priceUSD);
            if (sortVal === 'highLow') filtered.sort((a,b) => b.priceUSD - a.priceUSD);
            if (sortVal === 'rating') filtered.sort((a,b) => b.rating - a.rating);

            renderCatalog(filtered);
        }

        // Cart Logic
        function addToCart(pId) {
            const p = products.find(i => i.id === pId);
            if (!p) return;

            const existing = cart.find(i => i.id === pId);
            if (existing) {
                existing.qty += 1;
            } else {
                cart.push({ id: p.id, title: p.title, priceUSD: p.priceUSD, image: p.image, qty: 1 });
            }

            saveCart();
            updateCartUI();
            showToast(`Added ${p.title} to cart`);
        }

        function updateCartUI() {
            const list = document.getElementById('cartItemsList');
            const badge = document.getElementById('cartBadge');
            
            const totalQty = cart.reduce((sum, i) => sum + i.qty, 0);
            badge.innerText = totalQty;

            list.innerHTML = '';

            if (cart.length === 0) {
                list.innerHTML = `<p class="text-center text-slate-500 text-xs py-8 font-mono">Your shopping cart is empty.</p>`;

                cart.forEach(item => {
